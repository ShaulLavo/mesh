package main

import (
	"errors"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"runtime"
	"runtime/pprof"
	"sync"
	"time"
)

var benchCreateProfile = func(name string) (io.WriteCloser, error) { return os.Create(name) }

// pprof's asynchronous CPU writer discards write errors; retain them until Stop returns.
type benchCPUWriter struct {
	io.WriteCloser
	mu  sync.Mutex
	err error
}

func (w *benchCPUWriter) Write(data []byte) (int, error) {
	n, err := w.WriteCloser.Write(data)
	if n != len(data) && err == nil {
		err = io.ErrShortWrite
	}
	w.mu.Lock()
	w.err = errors.Join(w.err, err)
	w.mu.Unlock()
	return n, err
}

func (w *benchCPUWriter) closeChecked() error {
	err := w.Close()
	w.mu.Lock()
	defer w.mu.Unlock()
	return errors.Join(w.err, err)
}

// The overlay is confined to scratch builds. Baselines use an ordinary binary.
func benchProfile() func() {
	dir := os.Getenv("MESH_BENCH_PROFILE_DIR")
	if dir == "" {
		return func() {}
	}
	role := "cli"
	if len(os.Args) > 1 {
		role = filepath.Base(os.Args[1])
	}
	prefix := filepath.Join(dir, fmt.Sprintf("%s-%d", role, os.Getpid()))
	if err := os.Remove(prefix + ".done"); err != nil && !os.IsNotExist(err) {
		panic(err)
	}
	file, err := benchCreateProfile(prefix + ".cpu.pprof")
	if err != nil {
		panic(err)
	}
	cpu := &benchCPUWriter{WriteCloser: file}
	if err := pprof.StartCPUProfile(cpu); err != nil {
		panic(errors.Join(err, cpu.closeChecked()))
	}
	var once sync.Once
	finish := func() {
		once.Do(func() {
			pprof.StopCPUProfile()
			if err := cpu.closeChecked(); err != nil {
				panic(err)
			}
			runtime.GC()
			for _, name := range []string{"heap", "allocs"} {
				file, err := benchCreateProfile(prefix + "." + name + ".pprof")
				if err != nil {
					panic(err)
				}
				writeErr := pprof.Lookup(name).WriteTo(file, 0)
				if err := errors.Join(writeErr, file.Close()); err != nil {
					panic(err)
				}
			}
			if err := os.WriteFile(prefix+".done", nil, 0o600); err != nil {
				panic(err)
			}
		})
	}
	time.AfterFunc(5*time.Second, finish)
	return finish
}
